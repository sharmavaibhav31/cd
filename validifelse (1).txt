pgm.l:
    %{
    #include "y.tab.h"
    %}

    %%
    "if"      { return IF; }
    "else"    { return ELSE; }
    "="       { return ASSIGN; }
    ";"       { return SEMICOLON; }
    [0-9]+    { return NUM; }
    "=="|"!="|"<="|">="|"<"|">" { return RELOP; }
    [a-zA-Z_][a-zA-Z0-9_]*      { return ID; }
    [ \t\n\r] ; 
    .         { return yytext[0]; } 
    %%
    int yywrap() { return 1; }

pgm.y:
    %{
    #include <stdio.h>
    #include <stdlib.h>
    int yylex();
    void yyerror(const char *s);
    %}

    %token IF ELSE ID NUM RELOP ASSIGN SEMICOLON

    %%

    stmt: if_else { printf("Valid Syntax\n"); exit(0); };

    if_else: IF '(' cond ')' block

        | IF '(' cond ')' block ELSE block
        ;

    cond: ID RELOP ID 
        | ID RELOP NUM 
        | ID 
        ;

    block: '{' s '}' 

        | s 
        ;

    s: ID ASSIGN ID SEMICOLON 
    | ID ASSIGN NUM SEMICOLON
    | if_else 
    ;

    %%
    void main() {
        printf("Enter code: ");
        yyparse();
    }

    void yyerror(const char *s) { printf("Invalid\n"); }
